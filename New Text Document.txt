-- Tables
CREATE TABLE Ucenik
(
    Ucid INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
    Ime NVARCHAR(20) NOT NULL,
    Prezime NVARCHAR(20) NOT NULL,
    Dtmrod DATE,
    Imegrada NVARCHAR(20) NOT NULL
);
GO

CREATE TABLE Predmet
(
    PRid INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
    Naziv VARCHAR(20) NOT NULL,
    Prof VARCHAR(20) NOT NULL
);
GO

CREATE TABLE Ocjenivanje
(
    Ucenikid INT NOT NULL,
    Predmetid INT NOT NULL,
    Ocjena INT NOT NULL,
    Datum DATE,
    FOREIGN KEY (Ucenikid) REFERENCES Ucenik(Ucid),
    FOREIGN KEY (Predmetid) REFERENCES Predmet(PRid)
);
GO

-- 7. Sve ocjene iz Matematike
SELECT U.Ime, U.Prezime, O.Ocjena
FROM Ucenik U
INNER JOIN Ocjenivanje O ON U.Ucid = O.Ucenikid
INNER JOIN Predmet P ON O.Predmetid = P.PRid
WHERE P.Naziv = 'Matematika';
GO

-- 8. Svi ucenici koji imaju ocjenu 5 i naziv predmeta
SELECT U.Ime, U.Prezime, P.Naziv AS Predmet, O.Ocjena
FROM Ucenik U
INNER JOIN Ocjenivanje O ON U.Ucid = O.Ucenikid
INNER JOIN Predmet P ON O.Predmetid = P.PRid
WHERE O.Ocjena = 5;
GO

-- 9. Sve ocjene koje je dao odredjeni profesor
-- Primjer: Milan Milic
SELECT U.Ime, U.Prezime, P.Naziv AS Predmet,
       O.Ocjena, O.Datum
FROM Ucenik U
INNER JOIN Ocjenivanje O ON U.Ucid = O.Ucenikid
INNER JOIN Predmet P ON O.Predmetid = P.PRid
WHERE P.Prof = 'Milan Milic';
GO

-- 10. Ucenici koji imaju ocjenu 5 iz Matematike ili Hemije
SELECT DISTINCT U.Ime, U.Prezime
FROM Ucenik U
INNER JOIN Ocjenivanje O ON U.Ucid = O.Ucenikid
INNER JOIN Predmet P ON O.Predmetid = P.PRid
WHERE O.Ocjena = 5
AND P.Naziv IN ('Matematika', 'Hemija');
GO
